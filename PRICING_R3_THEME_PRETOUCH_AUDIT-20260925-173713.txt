     2	 * Generate a CSS color with alpha transparency.
     3	 * Supports hex (#RRGGBB / #RGB), rgb(), rgba(), CSS variables, and named colors.
     4	 */
     5	export function alpha(color: string, opacity: number): string {
     6	  const a = Math.max(0, Math.min(1, opacity));
     7	
     8	  // Hex: #RRGGBB or #RGB
     9	  if (color.startsWith("#")) {
    10	    const hex = color.replace("#", "");
    11	    let r: number, g: number, b: number;
    12	    if (hex.length === 6) {
    13	      r = parseInt(hex.slice(0, 2), 16);
    14	      g = parseInt(hex.slice(2, 4), 16);
    15	      b = parseInt(hex.slice(4, 6), 16);
    16	    } else if (hex.length === 3) {
    17	      r = parseInt(hex[0] + hex[0], 16);
    18	      g = parseInt(hex[1] + hex[1], 16);
    19	      b = parseInt(hex[2] + hex[2], 16);
    20	    } else {
    21	      return color; // malformed hex — pass through
    22	    }
    23	    return `rgba(${r}, ${g}, ${b}, ${a})`;
    24	  }
    25	
    26	  // rgb(r, g, b) → rgba(r, g, b, a)
    27	  if (color.startsWith("rgb(") && !color.startsWith("rgba(")) {
    28	    return color.replace("rgb(", "rgba(").replace(")", `, ${a})`);
    29	  }
    30	
    31	  // rgba(r, g, b, oldAlpha) → replace old alpha
    32	  if (color.startsWith("rgba(")) {
    33	    return color.replace(/,\s*[\d.]+\s*\)$/, `, ${a})`);
    34	  }
    35	
    36	  // CSS variables or named colors → color-mix fallback
    37	  return `color-mix(in srgb, ${color} ${Math.round(a * 100)}%, transparent)`;
    38	}
